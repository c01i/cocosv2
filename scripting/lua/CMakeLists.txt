cmake_minimum_required(VERSION 3.22)
project(cocos_lua515 C)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/lua/sources.txt" LUA515_FILES)
list(TRANSFORM LUA515_FILES PREPEND "${CMAKE_CURRENT_SOURCE_DIR}/lua/")
add_library(lua515 STATIC ${LUA515_FILES})
set_target_properties(lua515 PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 99)
target_include_directories(lua515 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/lua")
if(UNIX)
    target_compile_definitions(lua515 PRIVATE LUA_USE_POSIX LUA_USE_DLOPEN)
    target_link_libraries(lua515 PUBLIC m ${CMAKE_DL_LIBS})
endif()
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(lua515_check tests/runtime_check.c)
        target_link_libraries(lua515_check PRIVATE lua515)
        add_test(NAME lua515_runtime COMMAND lua515_check "${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime.lua")
    endif()
endif()
